cmake_minimum_required(VERSION 3.22.1)
project(ARUT_HARDT_ENGINE VERSION 0.1.0 DESCRIPTION "Independent Android-first open-world game engine" LANGUAGES C CXX)
option(ARUT_BUILD_TESTS "Build ARUT-HARDT tests" ON)
option(ARUT_BUILD_RUNTIME "Build the headless bootstrap executable" ON)
add_library(arut_core STATIC engine/core/src/version.cpp)
add_library(arut_animation STATIC engine/animation/src/runtime.cpp engine/animation/src/motion_matching.cpp)
target_include_directories(arut_animation PUBLIC engine/animation/include engine/core/include)
target_link_libraries(arut_animation PUBLIC arut_core)
target_compile_features(arut_animation PUBLIC cxx_std_20)
set_target_properties(arut_animation PROPERTIES CXX_EXTENSIONS OFF POSITION_INDEPENDENT_CODE ON)
if(MSVC)
 target_compile_options(arut_animation PRIVATE /W4 /permissive-)
else()
 target_compile_options(arut_animation PRIVATE -Wall -Wextra -Wpedantic)
endif()
add_library(arut_rendering STATIC engine/rendering/src/image_decode.cpp engine/rendering/src/environment.cpp engine/rendering/src/brdf_lut.cpp)
target_include_directories(arut_rendering PUBLIC engine/rendering/include PRIVATE third_party/stb)
target_compile_features(arut_rendering PUBLIC cxx_std_20)
set_target_properties(arut_rendering PROPERTIES CXX_EXTENSIONS OFF POSITION_INDEPENDENT_CODE ON)
if(MSVC)
 target_compile_options(arut_rendering PRIVATE /W4 /permissive-)
else()
 target_compile_options(arut_rendering PRIVATE -Wall -Wextra -Wpedantic)
endif()
target_include_directories(arut_core PUBLIC engine/core/include)
target_compile_features(arut_core PUBLIC cxx_std_20)
set_target_properties(arut_core PROPERTIES CXX_EXTENSIONS OFF POSITION_INDEPENDENT_CODE ON)
if(MSVC)
 target_compile_options(arut_core PRIVATE /W4 /permissive-)
else()
 target_compile_options(arut_core PRIVATE -Wall -Wextra -Wpedantic)
endif()

if(ANDROID)
 set(ARUT_RUST_STATICLIB "${CMAKE_SOURCE_DIR}/target/aarch64-linux-android/release/libarut_tasks.a")
 if(NOT EXISTS "${ARUT_RUST_STATICLIB}")
  message(FATAL_ERROR "Missing Rust ARM64 static library. Build Cargo target aarch64-linux-android first.")
 endif()
 add_library(arut_tasks STATIC IMPORTED GLOBAL)
 set_target_properties(arut_tasks PROPERTIES IMPORTED_LOCATION "${ARUT_RUST_STATICLIB}")
 set(ANDROID_NATIVE_APP_GLUE "${ANDROID_NDK}/sources/android/native_app_glue/android_native_app_glue.c")
 if(NOT EXISTS "${ANDROID_NATIVE_APP_GLUE}")
  message(FATAL_ERROR "Android NDK native_app_glue.c was not found.")
 endif()
 add_library(arut_android SHARED engine/platform/android/src/android_main.cpp engine/rendering/src/android_renderer.cpp "${ANDROID_NATIVE_APP_GLUE}")
 target_include_directories(arut_android PRIVATE engine/platform/android/include engine/rendering/include "${ANDROID_NDK}/sources/android/native_app_glue")
 target_compile_features(arut_android PRIVATE cxx_std_20)
 target_compile_definitions(arut_android PRIVATE ARUT_ANDROID_PLATFORM=1 VK_USE_PLATFORM_ANDROID_KHR=1)
 target_link_libraries(arut_android PRIVATE arut_core arut_animation arut_rendering arut_tasks android log vulkan dl m)
 set_target_properties(arut_android PROPERTIES CXX_EXTENSIONS OFF OUTPUT_NAME "arut_android")
endif()
